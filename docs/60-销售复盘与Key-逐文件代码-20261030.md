## A7. `config/TokenAwareChatModel.java`（完整替换）

```java
package com.aiextract.config;

import com.aiextract.service.TokenUsageService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.metadata.Usage;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.ai.chat.model.ChatResponse;
import org.springframework.ai.chat.prompt.Prompt;
import reactor.core.publisher.Flux;

import java.util.UUID;
import java.util.concurrent.atomic.AtomicReference;

@Slf4j
public class TokenAwareChatModel implements ChatModel {

    private static final int PROMPT_WARN_THRESHOLD = 10_000;
    private static final int CHARS_PER_TOKEN = 2;

    private final ChatModel delegate;
    private final TokenUsageService tokenUsageService;
    private final String modelName;
    /** null = 平台 Key；非 null = 企业 Key 代调用，用量强制归属该企业 */
    private final UUID companyId;

    public TokenAwareChatModel(ChatModel delegate, TokenUsageService tokenUsageService, String modelName) {
        this(delegate, tokenUsageService, modelName, null);
    }

    public TokenAwareChatModel(ChatModel delegate, TokenUsageService tokenUsageService,
                               String modelName, UUID companyId) {
        this.delegate = delegate;
        this.tokenUsageService = tokenUsageService;
        this.modelName = modelName;
        this.companyId = companyId;
    }

    @Override
    public ChatResponse call(Prompt prompt) {
        int promptChars = countPromptChars(prompt);
        auditPromptSize(promptChars);
        ChatResponse response = delegate.call(prompt);
        Usage usage = response.getMetadata() != null ? response.getMetadata().getUsage() : null;
        String responseContent = response.getResult() != null && response.getResult().getOutput() != null
                ? response.getResult().getOutput().getContent() : "";
        int completionChars = responseContent != null ? responseContent.length() : 0;

        int inputTokens;
        int outputTokens;
        if (usage != null && usage.getTotalTokens() > 0) {
            inputTokens = (int) (long) usage.getPromptTokens();
            outputTokens = (int) (long) usage.getGenerationTokens();
        } else {
            inputTokens = promptChars / CHARS_PER_TOKEN;
            outputTokens = completionChars / CHARS_PER_TOKEN;
        }
        log.info("LLM call() done promptChars={} completionChars={} inputTokens={} outputTokens={} model={}",
                promptChars, completionChars, inputTokens, outputTokens, modelName);
        UUID effectiveCompanyId = companyId != null ? companyId : TokenContext.getCompanyId();
        logUsage(TokenContext.get(), effectiveCompanyId, inputTokens, outputTokens, promptChars, completionChars);
        return response;
    }

    @Override
    public Flux<ChatResponse> stream(Prompt prompt) {
        int promptChars = countPromptChars(prompt);
        auditPromptSize(promptChars);
        // 请求线程捕获：doOnComplete 在 Reactor 线程执行，TokenContext 已被清空
        UUID userId = TokenContext.get();
        UUID effectiveCompanyId = companyId != null ? companyId : TokenContext.getCompanyId();

        AtomicReference<Usage> lastUsage = new AtomicReference<>();
        StringBuilder fullResponse = new StringBuilder();

        return delegate.stream(prompt)
                .doOnNext(resp -> {
                    if (resp.getResult() != null && resp.getResult().getOutput() != null) {
                        String content = resp.getResult().getOutput().getContent();
                        if (content != null) fullResponse.append(content);
                    }
                    Usage usage = resp.getMetadata() != null ? resp.getMetadata().getUsage() : null;
                    if (usage != null && usage.getTotalTokens() > 0) lastUsage.set(usage);
                })
                .doOnComplete(() -> {
                    int completionChars = fullResponse.length();
                    Usage usage = lastUsage.get();
                    int inputTokens;
                    int outputTokens;
                    if (usage != null && usage.getTotalTokens() > 0 && usage.getPromptTokens() > 0) {
                        inputTokens = (int) (long) usage.getPromptTokens();
                        outputTokens = (int) (long) usage.getGenerationTokens();
                    } else {
                        inputTokens = promptChars / CHARS_PER_TOKEN;
                        outputTokens = completionChars / CHARS_PER_TOKEN;
                    }
                    log.info("LLM stream() done promptChars={} completionChars={} inputTokens={} outputTokens={} model={}",
                            promptChars, completionChars, inputTokens, outputTokens, modelName);
                    logUsage(userId, effectiveCompanyId, inputTokens, outputTokens, promptChars, completionChars);
                })
                .doOnError(err -> {
                    int completionChars = fullResponse.length();
                    int inputTokens = promptChars / CHARS_PER_TOKEN;
                    int outputTokens = completionChars / CHARS_PER_TOKEN;
                    log.warn("LLM stream() error promptChars={} completionChars={} err={}",
                            promptChars, completionChars, err.getMessage());
                    logUsage(userId, effectiveCompanyId, inputTokens, outputTokens, promptChars, completionChars);
                });
    }

    private int countPromptChars(Prompt prompt) {
        if (prompt == null || prompt.getInstructions() == null) return 0;
        return prompt.getInstructions().stream()
                .mapToInt(msg -> msg.getContent() != null ? msg.getContent().length() : 0)
                .sum();
    }

    private void auditPromptSize(int promptChars) {
        if (promptChars > PROMPT_WARN_THRESHOLD) {
            log.warn("⚠️ 大 Prompt 检测 promptChars={} > {} 阈值, model={}",
                    promptChars, PROMPT_WARN_THRESHOLD, modelName);
        }
        log.debug("LLM REQ promptChars={} model={}", promptChars, modelName);
    }

    private void logUsage(UUID userId, UUID effectiveCompanyId, int inputTokens, int outputTokens,
                          int promptChars, int completionChars) {
        tokenUsageService.log(userId, effectiveCompanyId, "CHAT", modelName,
                inputTokens, outputTokens, promptChars, completionChars);
    }
}
```

```java
package com.aiextract.service;

import com.aiextract.config.PartnerCrypto;
import com.aiextract.config.TokenContext;
import com.aiextract.exception.BusinessException;
import com.aiextract.model.CompanyAiConfig;
import com.aiextract.repository.CompanyAiConfigRepository;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

@Slf4j
@Component
public class ChatClientResolver {

    private final ChatClient platformClient;
    private final CompanyAiConfigRepository repository;
    private final PartnerCrypto crypto;
    private final CompanyChatClientFactory factory;
    private final ConcurrentHashMap<UUID, CacheEntry> cache = new ConcurrentHashMap<>();

    public ChatClientResolver(@Qualifier("platformChatClient") ChatClient platformClient,
                              CompanyAiConfigRepository repository,
                              PartnerCrypto crypto,
                              CompanyChatClientFactory factory) {
        this.platformClient = platformClient;
        this.repository = repository;
        this.crypto = crypto;
        this.factory = factory;
    }

    public ChatClient current() {
        UUID companyId = TokenContext.getCompanyId();
        if (companyId == null) {
            return platformClient;
        }
        CompanyAiConfig cfg = repository.findByCompanyIdAndProvider(companyId, "deepseek").orElse(null);
        if (cfg == null) {
            return platformClient;
        }
        if (!"enabled".equals(cfg.getStatus())) {
            if (cfg.isFallbackEnabled()) return platformClient;
            throw new BusinessException(409, "AI_KEY_DISABLED", "企业 AI Key 已停用，请联系管理员");
        }
        CacheEntry entry = cache.get(companyId);
        if (entry != null && entry.updatedAt().equals(cfg.getUpdatedAt())) {
            return entry.client();
        }
        String apiKey;
        try {
            apiKey = crypto.decrypt(cfg.getApiKeyCipher());
        } catch (Exception e) {
            log.error("企业 AI Key 解密失败 companyId={}", companyId, e);
            if (cfg.isFallbackEnabled()) return platformClient;
            throw new BusinessException(409, "AI_KEY_DECRYPT_FAILED", "企业 AI Key 解密失败，请联系管理员");
        }
        ChatClient client = factory.create(companyId, cfg.getBaseUrl(), apiKey, cfg.getModelName());
        cache.put(companyId, new CacheEntry(client, cfg.getUpdatedAt()));
        return client;
    }

    public void evict(UUID companyId) {
        cache.remove(companyId);
    }

    private record CacheEntry(ChatClient client, LocalDateTime updatedAt) {}
}
```

```java
package com.aiextract.service;

import com.aiextract.config.TokenAwareChatModel;
```

```java
package com.aiextract.service;

import org.springframework.ai.chat.client.ChatClient;

import java.lang.reflect.Proxy;

public final class TenantChatClientProxy {

    private TenantChatClientProxy() {}

    public static ChatClient of(ChatClientResolver resolver) {
        return (ChatClient) Proxy.newProxyInstance(
            ChatClient.class.getClassLoader(),
            new Class[]{ChatClient.class},
            (proxy, method, args) -> {
                if (method.getDeclaringClass() == Object.class) {
                    return switch (method.getName()) {
                        case "toString" -> "TenantChatClientProxy";
                        case "hashCode" -> System.identityHashCode(proxy);
                        case "equals" -> proxy == args[0];
                        default -> null;
                    };
                }
                return method.invoke(resolver.current(), args);
            });
    }
}
```

```java
package com.aiextract.service;

import com.aiextract.config.PartnerCrypto;
import com.aiextract.config.TokenContext;
import com.aiextract.dto.*;
import com.aiextract.exception.BusinessException;
import com.aiextract.model.CompanyAiConfig;
import com.aiextract.repository.CompanyAiConfigRepository;
import com.aiextract.repository.TokenUsageLogRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.openai.OpenAiChatOptions;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.UUID;

@Slf4j
@Service
@RequiredArgsConstructor
public class CompanyAiConfigService {

    private final CompanyAiConfigRepository repository;
    private final PartnerCrypto crypto;
    private final ChatClientResolver resolver;
    private final SalesAuditService audit;
    private final TokenUsageLogRepository usageRepository;
    private final CompanyChatClientFactory factory;

    @Value("${spring.ai.openai.chat.options.model:deepseek-chat}")
    private String platformModelName;
    @Value("${spring.ai.openai.chat.options.base-url:https://api.deepseek.com}")
    private String platformBaseUrl;

    public CompanyAiConfigView get(UUID companyId) {
        CompanyAiConfig cfg = repository.findByCompanyIdAndProvider(companyId, "deepseek").orElse(null);
        TokenUsageLogRepository.CompanyUsageTotal usage =
            usageRepository.sumByCompanySince(companyId, LocalDate.now().withDayOfMonth(1));
        long input = usage == null || usage.getInputTokens() == null ? 0 : usage.getInputTokens();
        long output = usage == null || usage.getOutputTokens() == null ? 0 : usage.getOutputTokens();
        return new CompanyAiConfigView(
            cfg != null, "deepseek", cfg == null ? platformModelName : cfg.getModelName(),
            cfg == null ? null : cfg.getBaseUrl(),
            cfg == null ? null : cfg.getKeyLast4(),
            cfg == null ? "disabled" : cfg.getStatus(),
            cfg == null || cfg.isFallbackEnabled(),
            true, input, output);
    }

    @Transactional
    public CompanyAiConfigView update(UUID companyId, CompanyAiConfigRequest req) {
        validateBaseUrl(req.baseUrl());
        CompanyAiConfig cfg = repository.findByCompanyIdAndProvider(companyId, "deepseek")
            .orElseGet(() -> newConfig(companyId));
        cfg.setBaseUrl(req.baseUrl().trim());
        if (req.apiKey() != null && !req.apiKey().isBlank()) {
            String key = req.apiKey().trim();
            validateApiKey(key);
            cfg.setApiKeyCipher(crypto.encrypt(key));
            cfg.setKeyLast4(last4(key));
        }
        if (cfg.getApiKeyCipher() == null) {
            throw new BusinessException(400, "首次配置必须填写 API Key");
        }
        cfg.setFallbackEnabled(req.fallbackEnabled() == null || req.fallbackEnabled());
        cfg.setStatus("enabled");
        cfg.setUpdatedBy(TokenContext.get());
        cfg.setUpdatedAt(LocalDateTime.now());
        repository.save(cfg);
        resolver.evict(companyId);
        audit.record("ai.model.config.update", "company_ai_config", cfg.getId(),
            Map.of("baseUrl", cfg.getBaseUrl(), "keyLast4", cfg.getKeyLast4(),
                   "fallbackEnabled", cfg.isFallbackEnabled()));
        return get(companyId);
    }

    public CompanyAiTestResult test(UUID companyId, CompanyAiTestRequest req) {
        String plain = resolvePlainKey(companyId, req == null ? null : req.apiKey());
        String url = req == null || req.baseUrl() == null || req.baseUrl().isBlank()
            ? resolveBaseUrl(companyId) : req.baseUrl().trim();
        validateBaseUrl(url);
        long t0 = System.currentTimeMillis();
        try {
            ChatClient client = factory.create(companyId, url, plain,
                resolveModelName(companyId));
            client.prompt().user("ping")
                .options(OpenAiChatOptions.builder().maxTokens(1).build())
                .call().content();
            audit.record("ai.model.test", "company_ai_config", null,
                Map.of("ok", true, "baseUrl", url));
            return new CompanyAiTestResult(true, System.currentTimeMillis() - t0, null, "连接成功");
        } catch (Exception e) {
            log.warn("企业 AI Key 连通性测试失败 companyId={} baseUrl={}", companyId, url, e);
            audit.record("ai.model.test", "company_ai_config", null,
                Map.of("ok", false, "baseUrl", url, "error", String.valueOf(e.getMessage())));
            return new CompanyAiTestResult(false, System.currentTimeMillis() - t0,
                mapErrorCode(e), mapErrorMessage(e));
        }
    }

    @Transactional
    public void rotate(UUID companyId, String newKey) {
        validateApiKey(newKey);
        CompanyAiConfig cfg = require(companyId);
        String previousKeyLast4 = cfg.getKeyLast4();
        cfg.setApiKeyCipher(crypto.encrypt(newKey.trim()));
        cfg.setKeyLast4(last4(newKey.trim()));
        cfg.setUpdatedBy(TokenContext.get());
        cfg.setUpdatedAt(LocalDateTime.now());
        repository.save(cfg);
        resolver.evict(companyId);
        // 旧 Key 不再落库：审计留末 4 位与废弃标记即可
        audit.record("ai.model.rotate", "company_ai_config", cfg.getId(),
            Map.of("keyLast4", cfg.getKeyLast4(),
                   "previousKeyLast4", previousKeyLast4 == null ? "" : previousKeyLast4,
                   "previousKeyRetained", false));
    }

    @Transactional
    public void disable(UUID companyId) {
        CompanyAiConfig cfg = require(companyId);
        cfg.setStatus("disabled");
        cfg.setUpdatedBy(TokenContext.get());
        cfg.setUpdatedAt(LocalDateTime.now());
        repository.save(cfg);
        resolver.evict(companyId);
        audit.record("ai.model.disable", "company_ai_config", cfg.getId(),
            Map.of("fallbackEnabled", cfg.isFallbackEnabled()));
    }

    private CompanyAiConfig require(UUID companyId) {
        return repository.findByCompanyIdAndProvider(companyId, "deepseek")
            .orElseThrow(() -> new BusinessException(404, "企业尚未配置 AI Key"));
    }

    private CompanyAiConfig newConfig(UUID companyId) {
        LocalDateTime now = LocalDateTime.now();
        return CompanyAiConfig.builder()
            .id(UUID.randomUUID()).companyId(companyId).provider("deepseek")
            .baseUrl(platformBaseUrl).modelName(platformModelName)
            .status("disabled").fallbackEnabled(true)
            .createdBy(TokenContext.get()).createdAt(now).updatedAt(now).build();
    }

    private String resolvePlainKey(UUID companyId, String inputKey) {
        if (inputKey != null && !inputKey.isBlank()) return inputKey.trim();
        CompanyAiConfig cfg = repository.findByCompanyIdAndProvider(companyId, "deepseek")
            .orElseThrow(() -> new BusinessException(400, "未填写 API Key，企业也尚未配置"));
        return crypto.decrypt(cfg.getApiKeyCipher());
    }

    private String resolveBaseUrl(UUID companyId) {
        return repository.findByCompanyIdAndProvider(companyId, "deepseek")
            .map(CompanyAiConfig::getBaseUrl).orElse(platformBaseUrl);
    }

    private String resolveModelName(UUID companyId) {
        return repository.findByCompanyIdAndProvider(companyId, "deepseek")
            .map(CompanyAiConfig::getModelName).orElse(platformModelName);
    }

    private void validateBaseUrl(String url) {
        if (url == null || url.isBlank()) throw new BusinessException(400, "Base URL 不能为空");
        String u = url.trim();
        boolean https = u.startsWith("https://");
        boolean local = u.startsWith("http://localhost") || u.startsWith("http://127.0.0.1");
        if (!https && !local) throw new BusinessException(400, "Base URL 必须以 https:// 开头");
        if (u.length() > 300) throw new BusinessException(400, "Base URL 过长");
    }

    private void validateApiKey(String key) {
        if (key == null || key.trim().length() < 8 || key.trim().length() > 300) {
            throw new BusinessException(400, "API Key 长度需在 8–300 之间");
        }
    }

    private String last4(String key) {
        String k = key.trim();
        return k.length() <= 4 ? k : k.substring(k.length() - 4);
    }

    private String mapErrorCode(Exception e) {
        String msg = String.valueOf(e.getMessage());
        if (msg.contains("401") || msg.toLowerCase().contains("unauthorized")) return "AI_KEY_INVALID";
        if (msg.contains("402") || msg.toLowerCase().contains("balance")) return "AI_KEY_BALANCE";
        if (msg.contains("429") || msg.toLowerCase().contains("rate limit")) return "AI_KEY_RATE_LIMIT";
        if (msg.toLowerCase().contains("timeout")) return "AI_KEY_TIMEOUT";
        return "AI_KEY_UNREACHABLE";
    }

    private String mapErrorMessage(Exception e) {
        return switch (mapErrorCode(e)) {
            case "AI_KEY_INVALID" -> "Key 无效或没有权限";
            case "AI_KEY_BALANCE" -> "账户余额不足";
            case "AI_KEY_RATE_LIMIT" -> "请求过于频繁或额度已用尽";
            case "AI_KEY_TIMEOUT" -> "连接超时";
            default -> "无法连接，请检查 Base URL 与网络";
        };
    }
}
```

## A19. W1 验收点

1. 未配置企业 Key：`get()` 返回 `configured=false`，调用任意聊天接口使用平台 Key，流程正常。
2. 保存企业 Key 后：`keyLast4` 正确，DB 只存密文；日志无明文。
3. 企业 Key 额度不足（402）或无效（401）：请求失败并返回对应错误，**不自动回退平台**。
4. 停用且 `fallbackEnabled=true`：回到平台 Key；`false`：409 `AI_KEY_DISABLED`。
5. 轮换只替换当前 Key，旧 Key 密文不落库；`admin_audit_log` 的 `ai.model.rotate` 记录旧 Key 末 4 位与 `previousKeyRetained=false`。
6. 培训 Worker / 素材萃取 / 评测 Worker 的 `TokenContext` 覆盖生效后，用量归属企业。

**已知取舍**：`POST /admin/company/ai-model/test` 会同步打一次 LLM（10 秒超时），是唯一在 HTTP 线程里调 AI 的接口。它是管理员手动诊断按钮、单次 `maxTokens=1`、限 COMPANY 管理员，不参与业务链路；若后续要严格遵守「HTTP 线程不调 AI」，改成 202 + 轮询 job 即可，接口形状不用变。

**本轮落地顺序**：先做企业 Key（A 部分），再做复盘（B/C）。企业 Key 的异步上下文覆盖，复盘 Worker 随 B 一起做；培训/素材/评测三处按 2.8 的 P0 清单在上线前补，避免静默用平台 Key。

### `service/ReviewSegmenter.java`

```java
@Component
public class ReviewSegmenter {

    private static final int MAX_SEGMENTS = 500;
    private static final int MAX_SEGMENT_CHARS = 2000;

    public record SegmentDraft(String speaker, String text) {}

    public List<SegmentDraft> split(String transcript) {
        List<SegmentDraft> out = new ArrayList<>();
        for (String raw : transcript.split("\\r?\\n")) {
            String line = raw.trim();
            if (line.isEmpty()) continue;
            for (String piece : splitLongLine(line, MAX_SEGMENT_CHARS)) {
                out.add(new SegmentDraft(detectSpeaker(piece), stripPrefix(piece)));
                if (out.size() >= MAX_SEGMENTS) return out;
            }
        }
        if (out.isEmpty()) {
            out.add(new SegmentDraft("unknown", transcript.trim()));
        }
        return out;
    }

    private List<String> splitLongLine(String line, int max) {
        List<String> parts = new ArrayList<>();
        int start = 0;
        while (start < line.length()) {
            int end = Math.min(start + max, line.length());
            parts.add(line.substring(start, end));
            start = end;
        }
        return parts;
    }

    private String detectSpeaker(String line) {
        if (line.startsWith("客户") || line.startsWith("对方")) return "customer";
        if (line.startsWith("销售") || line.startsWith("我")) return "sales";
        return "unknown";
    }

    private String stripPrefix(String line) {
        String s = line.replaceFirst("^(客户|对方|销售|我)\\s*[：:]\\s*", "");
        return s.isBlank() ? line : s;
    }
}
```

```java
@Service
@RequiredArgsConstructor
public class SalesReviewService {

    @Transactional
    public ReviewDetailView create(ReviewCreateRequest req) {
        UUID companyId = access.requireCompanyId();
        String transcript = normalizeText(req.transcript());
        if (transcript.length() > maxTranscriptChars) {
            throw new BusinessException(400, "对话文本不能超过 " + maxTranscriptChars + " 字");
        }
        Skill coach = skillRepository.findById(req.coachSkillId())
            .filter(s -> "individual".equals(s.getType()))
            .filter(s -> "published".equals(s.getStatus()))
            .orElseThrow(() -> new BusinessException(404, "教练分身不存在或不可用"));
        Space space = spaceRepository.findById(coach.getSpaceId())
            .orElseThrow(() -> new BusinessException(404, "教练分身归属异常"));
        UUID coachCompanyId = resolveSkillCompanyId(coach, space);
        if (!companyId.equals(coachCompanyId)) {
            throw new BusinessException(404, "教练分身不存在或不可用");
        }

        List<ReviewSegmenter.SegmentDraft> drafts = segmenter.split(transcript);
        LocalDateTime now = LocalDateTime.now();
        ReviewSession session = ReviewSession.builder()
            .id(UUID.randomUUID()).companyId(companyId).reviewerId(TokenContext.get())
            .coachSkillId(coach.getId()).coachOwnerId(space.getUserId())
            .coachName(coach.getDisplayName() != null ? coach.getDisplayName() : coach.getOwnerName())
            .coachTitle(coach.getOwnerTitle())
            .customerName(trimToNull(req.customerName())).topic(trimToNull(req.topic()))
            .transcript(transcript).transcriptChars(transcript.length())
            .source("text").shareTranscriptWithCoach(Boolean.TRUE.equals(req.shareTranscriptWithCoach()))
            .status("active").createdAt(now).updatedAt(now).build();
        sessionRepository.save(session);

        List<ReviewSegment> segments = new ArrayList<>();
        int seq = 1;
        for (ReviewSegmenter.SegmentDraft d : drafts) {
            segments.add(ReviewSegment.builder()
                .id(UUID.randomUUID()).sessionId(session.getId()).seq(seq++)
                .speaker(d.speaker()).text(d.text()).charCount(d.text().length())
                .createdAt(now).build());
        }
        segmentRepository.saveAll(segments);

        audit.record("sales.review.create", "review_session", session.getId(),
            Map.of("coachSkillId", coach.getId().toString(), "segments", segments.size()));
        return detail(session.getId());
    }

    public Page<ReviewSessionView> list(int page, int size, String status, String scope) {
        UUID companyId = access.requireCompanyId();
        boolean companyScope = "company".equals(scope);
        if (companyScope && !access.canViewAll()) {
            throw new BusinessException(403, "无全公司查看权限");
        }
        Pageable p = PageRequest.of(Math.max(page, 1) - 1, Math.min(Math.max(size, 1), 100));
        UUID me = TokenContext.get();
        String st = trimToNull(status);
        Page<ReviewSession> rows = companyScope
            ? (st == null
                ? sessionRepository.findByCompanyIdOrderByCreatedAtDesc(companyId, p)
                : sessionRepository.findByCompanyIdAndStatusOrderByCreatedAtDesc(companyId, st, p))
            : (st == null
                ? sessionRepository.findByCompanyIdAndReviewerIdOrderByCreatedAtDesc(companyId, me, p)
                : sessionRepository.findByCompanyIdAndReviewerIdAndStatusOrderByCreatedAtDesc(companyId, me, st, p));

        List<UUID> reviewerIds = rows.getContent().stream().map(ReviewSession::getReviewerId).distinct().toList();
        Map<UUID, String> names = loadNames(reviewerIds);
        List<UUID> sessionIds = rows.getContent().stream().map(ReviewSession::getId).toList();
        Map<UUID, ReviewReportView> latestReports = latestReports(sessionIds);
        return rows.map(s -> toView(s, names.get(s.getReviewerId()), null,
                latestReports.get(s.getId())));
    }

    public ReviewDetailView detail(UUID id) {
        ReviewSession s = access.requireSession(id);
        List<ReviewSegmentView> segments = segmentRepository.findBySessionIdOrderBySeqAsc(id).stream()
            .map(x -> new ReviewSegmentView(x.getId(), x.getSeq(), x.getSpeaker(), x.getText()))
            .toList();
        ReviewReport latest = reportRepository.findFirstBySessionIdOrderByVersionDesc(id).orElse(null);
        long taskCount = taskRepository.countByReviewId(id);
        Lead lead = s.getLeadId() == null ? null
            : leadRepository.findById(s.getLeadId()).orElse(null);
        ReviewSessionView view = toView(s, loadNames(List.of(s.getReviewerId())).get(s.getReviewerId()),
```

```java
@Service
@RequiredArgsConstructor
public class SalesReviewReportService {

    private final ReviewReportRepository reportRepository;
    private final ReviewSessionRepository sessionRepository;
    private final TrainingNotificationService notificationService;
    private final SalesAuditService audit;

    @Value("${app.sales.review.max-attempts:3}")
    private int maxAttempts;
    @Value("${app.sales.review.lease-minutes:10}")
    private int leaseMinutes;

    @Transactional
    public List<UUID> claimBatch(String owner, int limit) {
        List<UUID> ids = reportRepository.lockClaimableIds(limit, maxAttempts);
        LocalDateTime expires = LocalDateTime.now().plusMinutes(leaseMinutes);
        return ids.stream()
            .filter(id -> reportRepository.claim(id, owner, expires, maxAttempts) == 1)
            .toList();
    }

    @Transactional
    public void complete(UUID id, String owner, String summary, String payload, String model) {
        int updated = reportRepository.complete(id, owner, summary, payload, model);
        if (updated == 0) return;
        ReviewReport r = reportRepository.findById(id).orElseThrow();
        audit.record("sales.report.ready", "review_report", id,
            Map.of("version", r.getVersion(), "model", model));
        sessionRepository.findById(r.getSessionId()).ifPresent(s -> {
            if ("active".equals(s.getStatus())) {
                s.setStatus("report_ready");
                s.setUpdatedAt(LocalDateTime.now());
                sessionRepository.save(s);
            }
            notificationService.notifyUser(s.getReviewerId(), s.getCompanyId(),
                "sales_review_ready", "复盘报告已生成",
                "「" + (s.getCustomerName() == null ? "本次对话" : s.getCustomerName()) + "」的复盘已生成，点击查看。",
                "/sales/reviews/" + s.getId(), "review_report", r.getId(), "sales:report:" + r.getId());
        });
    }

    @Transactional
    public void fail(UUID id, String owner, String error) {
        ReviewReport r = reportRepository.findById(id).orElseThrow();
        boolean exhausted = r.getAttempts() >= maxAttempts;
        reportRepository.finishFailed(id, owner, exhausted ? "failed" : "queued",
            error == null ? "未知错误" : truncate(error, 1000));
        if (exhausted) {
            sessionRepository.findById(r.getSessionId()).ifPresent(s ->
                notificationService.notifyUser(s.getReviewerId(), s.getCompanyId(),
                    "sales_review_failed", "复盘报告生成失败",
                    "「" + (s.getCustomerName() == null ? "本次对话" : s.getCustomerName())
                        + "」的复盘连续失败，请稍后重试或联系管理员。",
                    "/sales/reviews/" + s.getId(), "review_report", id,
                    "sales:report-failed:" + id));
        }
    }

    private String truncate(String s, int max) {
        return s.length() <= max ? s : s.substring(0, max);
    }
```

```java
public class SalesReviewReportGenerator {

    private final ReviewReportRepository reportRepository;
    private final ReviewSessionRepository sessionRepository;
    private final ReviewSegmentRepository segmentRepository;
    private final SkillRepository skillRepository;
    private final GrainRetriever grainRetriever;
    private final SalesReviewPromptBuilder promptBuilder;
    private final SalesReviewReportParser parser;
    private final SalesReviewReportService reportService;
    private final ChatStreamAdapter chatStreamAdapter;

    public void generate(UUID reportId, String owner) {
        ReviewReport report = reportRepository.findById(reportId).orElseThrow();
        if (!"generating".equals(report.getStatus()) || !owner.equals(report.getLeaseOwner())) return;
        ReviewSession session = sessionRepository.findById(report.getSessionId()).orElseThrow();
        List<ReviewSegment> segments = segmentRepository.findBySessionIdOrderBySeqAsc(session.getId());
        TokenContext.callWith(report.getCreatedBy(), report.getCompanyId(), () -> {
            List<GrainRetriever.GrainResult> grains = retrieveGrains(session, segments);
            SalesReviewPromptBuilder.Prompt prompt = promptBuilder.buildReport(session, segments, grains);
            String content = chatStreamAdapter.chat(prompt.system(), prompt.user());
            if (content == null || content.isBlank()) {
                throw new IllegalStateException("模型返回为空");
            }
            Set<UUID> segmentIds = segments.stream().map(ReviewSegment::getId).collect(Collectors.toSet());
            SalesReviewReportParser.ParsedReport parsed = parser.parse(content, segmentIds,
                segments.stream().collect(Collectors.toMap(ReviewSegment::getId, ReviewSegment::getText)));
            reportService.complete(report.getId(), owner, parsed.summary(), parsed.payloadJson(),
                report.getModelName() == null ? "deepseek-chat" : report.getModelName());
            return null;
        });
    }

    private List<GrainRetriever.GrainResult> retrieveGrains(ReviewSession session, List<ReviewSegment> segments) {
        Skill coach = skillRepository.findById(session.getCoachSkillId()).orElse(null);
        if (coach == null || coach.getSpaceId() == null) return List.of();
        String query = buildQuery(session, segments);
        return grainRetriever.retrieveHybrid(query, coach.getSpaceId(), 8);
    }

    private String buildQuery(ReviewSession session, List<ReviewSegment> segments) {
        StringBuilder q = new StringBuilder();
        if (session.getCustomerName() != null) q.append("客户：").append(session.getCustomerName()).append(' ');
        if (session.getTopic() != null) q.append("主题：").append(session.getTopic()).append(' ');
        segments.stream().limit(10).forEach(s -> q.append(s.getText()).append('\n'));
        String text = q.toString();
        return text.length() > 500 ? text.substring(0, 500) : text;
    }
}
```

@Slf4j
@Component
@RequiredArgsConstructor
public class SalesReviewReportParser {

    private final ObjectMapper objectMapper;

    public record ParsedReport(String summary, String payloadJson) {}

    public ParsedReport parse(String raw, Set<UUID> segmentIds, Map<UUID, String> segmentTexts) {
        String json = stripFences(raw).trim();
        JsonNode root;
        try {
            root = objectMapper.readTree(json);
        } catch (Exception e) {
            throw new IllegalArgumentException("模型输出不是合法 JSON", e);
        }
        String summary = text(root, "summary");
        if (summary == null || summary.isBlank()) throw new IllegalArgumentException("缺少 summary");
        summary = summary.length() > 200 ? summary.substring(0, 200) : summary;

        List<Map<String, Object>> strengths = points(root, "strengths", segmentIds, segmentTexts, 8, true);
        List<Map<String, Object>> weaknesses = points(root, "weaknesses", segmentIds, segmentTexts, 8, true);
        List<Map<String, Object>> risks = points(root, "risks", segmentIds, segmentTexts, 8, true);
        List<Map<String, Object>> signals = signals(root, "customerSignals", segmentIds, segmentTexts, 8);
        List<Map<String, Object>> nextSteps = steps(root, "nextSteps", segmentIds, segmentTexts, 3);
        Map<String, Object> leadDraft = leadDraft(root);

        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("summary", summary);
        payload.put("strengths", strengths);
        payload.put("weaknesses", weaknesses);
        payload.put("risks", risks);
        payload.put("customerSignals", signals);
        payload.put("nextSteps", nextSteps);
        payload.put("leadDraft", leadDraft);
        try {
            return new ParsedReport(summary, objectMapper.writeValueAsString(payload));
        } catch (Exception e) {
            throw new IllegalStateException("报告序列化失败", e);
        }
    }

    private List<Map<String, Object>> points(JsonNode root, String field, Set<UUID> segmentIds,
                                             Map<UUID, String> texts, int max, boolean requireEvidence) {
        List<Map<String, Object>> out = new ArrayList<>();
        JsonNode arr = root.path(field);
        if (!arr.isArray()) return out;
        for (JsonNode n : arr) {
            if (out.size() >= max) break;
            String point = text(n, "point");
            if (point == null || point.isBlank()) continue;
            List<Map<String, Object>> evidence = evidence(n, segmentIds, texts);
            if (requireEvidence && evidence.isEmpty()) {
                throw new IllegalArgumentException(field + " 缺少有效引用");
            }
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("point", point);
            m.put("evidence", evidence);
            m.put("confidence", confidence(n));
            m.put("suggestion", text(n, "suggestion"));
            out.add(m);
        }
        return out;
    }

    private List<Map<String, Object>> signals(JsonNode root, String field, Set<UUID> segmentIds,
                                              Map<UUID, String> texts, int max) {
        List<Map<String, Object>> out = new ArrayList<>();
        JsonNode arr = root.path(field);
        if (!arr.isArray()) return out;
        for (JsonNode n : arr) {
            if (out.size() >= max) break;
            String name = text(n, "field");
            if (name == null || name.isBlank()) continue;
            String sourceType = text(n, "sourceType");
            if (!Set.of("customer_words", "sales_paraphrase", "ai_inference", "unknown").contains(sourceType)) {
                sourceType = "unknown";
            }
            String value = text(n, "value");
            if ("unknown".equals(sourceType)) value = null;
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("field", name);
            m.put("value", value);
            m.put("sourceType", sourceType);
            m.put("evidence", evidence(n, segmentIds, texts));
            m.put("confidence", confidence(n));
            out.add(m);
        }
        return out;
    }

    private List<Map<String, Object>> steps(JsonNode root, String field, Set<UUID> segmentIds,
                                            Map<UUID, String> texts, int max) {
        List<Map<String, Object>> out = new ArrayList<>();
        JsonNode arr = root.path(field);
        if (!arr.isArray()) return out;
        for (JsonNode n : arr) {
            if (out.size() >= max) break;
            String title = text(n, "title");
            if (title == null || title.isBlank()) continue;
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("title", title);
            m.put("channel", text(n, "channel"));
            m.put("dueInDays", n.path("dueInDays").isNumber() ? n.path("dueInDays").asInt() : null);
            m.put("content", text(n, "content"));
            m.put("evidence", evidence(n, segmentIds, texts));
            out.add(m);
        }
        return out;
    }

    private List<Map<String, Object>> evidence(JsonNode node, Set<UUID> segmentIds,
                                               Map<UUID, String> texts) {
        List<Map<String, Object>> out = new ArrayList<>();
        JsonNode arr = node.path("evidence");
        if (!arr.isArray()) return out;
        for (JsonNode e : arr) {
            String sid = text(e, "segmentId");
            if (sid == null) continue;
            UUID id;
            try {
                id = UUID.fromString(sid);
            } catch (Exception ex) {
                throw new IllegalArgumentException("引用段落 ID 非法：" + sid);
            }
            if (!segmentIds.contains(id)) {
                throw new IllegalArgumentException("引用段落不属于本次会话：" + sid);
            }
            String quote = text(e, "quote");
            if (quote != null && !quote.isBlank()) {
                String source = texts.getOrDefault(id, "");
                if (!squash(source).contains(squash(quote))) {
                    throw new IllegalArgumentException("引用原文与段落不匹配：" + sid);
                }
            }
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("segmentId", sid);
            if (quote != null) m.put("quote", quote);
            out.add(m);
        }
        return out;
    }

    private Map<String, Object> leadDraft(JsonNode root) {
        JsonNode d = root.path("leadDraft");
        Map<String, Object> out = new LinkedHashMap<>();
        out.put("name", d.path("name").isTextual() ? d.path("name").asText() : null);
        out.put("contactName", d.path("contactName").isTextual() ? d.path("contactName").asText() : null);
        out.put("contactPhone", d.path("contactPhone").isTextual() ? d.path("contactPhone").asText() : null);
        out.put("needSummary", d.path("needSummary").isTextual() ? d.path("needSummary").asText() : null);
        return out;
    }

    private String text(JsonNode node, String field) {
        JsonNode v = node.path(field);
        return v.isTextual() && !v.asText().isBlank() ? v.asText().trim() : null;
    }

    private String confidence(JsonNode node) {
        String c = text(node, "confidence");
        return Set.of("high", "medium", "low").contains(c) ? c : "low";
    }

    private String stripFences(String raw) {
        String s = raw == null ? "" : raw.trim();
        if (s.startsWith("```")) {
            int first = s.indexOf('\n');
            if (first > 0) s = s.substring(first + 1);
            int last = s.lastIndexOf("```");
            if (last >= 0) s = s.substring(0, last);
        }
        return s;
    }

    private String squash(String s) {
        return s == null ? "" : s.replaceAll("\\s+", "");
    }
}

```java
public class SalesReviewChatService {

    public List<ReviewMessageView> history(UUID sessionId) {
        access.requireSession(sessionId);
        List<ReviewMessage> rows = new ArrayList<>(
            messageRepository.findTop10BySessionIdOrderByCreatedAtDesc(sessionId));
        java.util.Collections.reverse(rows);
        return rows.stream()
            .map(m -> new ReviewMessageView(m.getId(), m.getRole(), m.getContent(), m.getCreatedAt()))
            .toList();
    }

    public Flux<ChatChunk> stream(UUID sessionId, String content) {
        ReviewSession session = access.requireSession(sessionId);
        long persisted = messageRepository.countBySessionIdAndRole(sessionId, "user");
        if (!rateLimiter.tryConsume(sessionId, persisted)) {
            throw BusinessException.conflict("SALES_REVIEW_CHAT_LIMIT", "本次复盘追问次数已达上限");
        }
        String system = promptBuilder.buildChatSystem(session,
            reportRepository.findFirstBySessionIdOrderByVersionDesc(sessionId).orElse(null));
        List<Map<String, String>> messages = promptBuilder.buildChatMessages(system, sessionId, content);

        StringBuilder answer = new StringBuilder();
        return chatStreamAdapter.chatStream(messages, Map.of())
            .map(ChatChunk::fromEventMap)
            .doOnNext(chunk -> {
                if ("content".equals(chunk.getType()) && chunk.getContent() != null) {
                    answer.append(chunk.getContent());
                }
            })
            .doOnComplete(() -> persistence.save(session, content, answer.toString()))
            .doOnError(e -> log.warn("复盘追问流式失败 sessionId={}", sessionId, e));
    }
}
```

### `service/SalesReviewRateLimiter.java`

```java
@Slf4j
@Component
@RequiredArgsConstructor
public class SalesReviewRateLimiter {

    private final StringRedisTemplate redisTemplate;

    @Value("${app.sales.review.chat.max-total-per-session:40}")
    private int maxTotal;
    @Value("${app.sales.review.chat.max-per-minute:10}")
    private int maxPerMinute;

    private static final DefaultRedisScript<Long> TOTAL_SCRIPT = new DefaultRedisScript<>(
        "local k = KEYS[1] "
            + "local init = tonumber(ARGV[1]) "
            + "local limit = tonumber(ARGV[2]) "
            + "if redis.call('EXISTS', k) == 0 and init > 0 then "
            + "  redis.call('SET', k, init, 'EX', 86400) "
            + "end "
            + "local n = redis.call('INCR', k) "
            + "if redis.call('TTL', k) < 0 then redis.call('EXPIRE', k, 86400) end "
            + "if n <= limit then return 1 else return 0 end",
        Long.class);

    private static final DefaultRedisScript<Long> MINUTE_SCRIPT = new DefaultRedisScript<>(
        "local n = redis.call('INCR', KEYS[1]) "
            + "if n == 1 then redis.call('EXPIRE', KEYS[1], 60) end "
            + "if n <= tonumber(ARGV[1]) then return 1 else return 0 end",
        Long.class);

    public boolean tryConsume(UUID sessionId, long persistedUserMessages) {
        try {
            Long total = redisTemplate.execute(TOTAL_SCRIPT,
                List.of("rate:sales-review-total:" + sessionId),
                String.valueOf(Math.max(persistedUserMessages, 0)), String.valueOf(maxTotal));
            if (total != null && total == 0L) return false;
            Long perMinute = redisTemplate.execute(MINUTE_SCRIPT,
                List.of("rate:sales-review-min:" + sessionId), String.valueOf(maxPerMinute));
            return perMinute == null || perMinute == 1L;
        } catch (Exception e) {
            log.warn("复盘追问限流 Redis 异常，回退 DB 计数 sessionId={}: {}", sessionId, e.getMessage());
            return persistedUserMessages < maxTotal;
        }
    }
}
```

## B8. W2 后端验收点

1. 建会话：跨企业教练 404、组织分身 404、非 published 404、3 万字以上 400。

7. 台账：只有教练本人能看到找过自己的记录；未授权原文 404；授权后可看段落。
8. 可见性：普通员工看不到同事的复盘；company_admin 能看全公司；跨企业一律 404。

```tsx
function localDateTime(d: Date) {
```

```tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import { ReviewMessage, streamMessage } from '@/lib/api/sales';

export function ReviewChat({
  reviewId, initial, disabled,
}: {
  reviewId: string;
  initial: ReviewMessage[];
  disabled?: boolean;
}) {
  const [messages, setMessages] = useState<ReviewMessage[]>(initial);
  const [input, setInput] = useState('');
  const [streaming, setStreaming] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const abortRef = useRef<AbortController | null>(null);

  useEffect(() => () => abortRef.current?.abort(), []);

  const send = () => {
    const text = input.trim();
    if (!text || streaming || disabled) return;
    setInput(''); setStreaming(true); setError(null);
    const assistantId = `tmp-${Date.now()}`;
    setMessages(prev => [...prev,
      { id: `tmp-u-${Date.now()}`, role: 'user', content: text, createdAt: new Date().toISOString() },
      { id: assistantId, role: 'assistant', content: '', createdAt: new Date().toISOString() },
    ]);
    let acc = '';
    abortRef.current = streamMessage(reviewId, text, {
      onChunk: chunk => {
        acc += chunk;
        setMessages(prev => prev.map(m => m.id === assistantId ? { ...m, content: acc } : m));
      },
      onDone: () => setStreaming(false),
      onError: msg => {
        setStreaming(false);
        setMessages(prev => prev.filter(m => m.id !== assistantId));
        setError(msg);
      },
    });
  };

  return (
    <section className="rounded border border-gray-200 p-4">
      <h3 className="mb-3 text-sm font-medium text-gray-900">继续问教练</h3>
      <div className="mb-3 max-h-80 space-y-3 overflow-auto">
        {messages.length === 0 && <p className="text-sm text-gray-500">还没有追问。可以问「这个客户最该先解决什么」。</p>}
        {messages.map(m => (
          <div key={m.id} className={m.role === 'user' ? 'text-right' : 'text-left'}>
            <div className={`inline-block max-w-[85%] whitespace-pre-wrap rounded px-3 py-2 text-sm ${
              m.role === 'user' ? 'bg-blue-600 text-white' : 'bg-gray-100 text-gray-900'
            }`}>{m.content || '…'}</div>
          </div>
        ))}
      </div>
      {error && <p className="mb-2 text-sm text-red-600">{error}</p>}
      <div className="flex gap-2">
        <input
          className="flex-1 rounded border border-gray-300 px-3 py-2 text-sm"
          value={input}
          disabled={disabled || streaming}
          onChange={e => setInput(e.target.value)}
          onKeyDown={e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); send(); } }}
```

```java
@Service
@RequiredArgsConstructor
public class SalesCoachSupplementService {

    private final SalesAccessService access;
    private final ReviewSessionRepository sessionRepository;
    private final ReviewReportRepository reportRepository;
    private final ReviewSegmentRepository segmentRepository;
    private final ReviewCoachRequestRepository requestRepository;
    private final ReviewCoachNoteRepository noteRepository;
    private final UserRepository userRepository;
    private final TrainingNotificationService notificationService;
    private final SalesAuditService audit;
    private final ObjectMapper objectMapper;

    /** 发起人请求教练补充；可同时授权原文 */
    @Transactional
    public CoachSupplementView request(UUID sessionId, CoachRequestCreateRequest req) {
        ReviewSession session = access.requireSession(sessionId);
        if (!session.getReviewerId().equals(TokenContext.get())) {
            throw new BusinessException(403, "只有复盘发起人能给教练发请求");
        }
        if ("closed".equals(session.getStatus())) {
            throw new BusinessException(409, "复盘已关闭");
        }
        LocalDateTime now = LocalDateTime.now();
        if (session.getCoachOwnerId().equals(TokenContext.get())) {
            // 自己复盘自己：不建请求，直接允许写补充
            return supplementView(sessionId);
        }
        ReviewCoachRequest r = requestRepository.findBySessionId(sessionId)
            .orElseGet(() -> ReviewCoachRequest.builder()
                .id(UUID.randomUUID()).companyId(session.getCompanyId())
                .sessionId(sessionId).coachUserId(session.getCoachOwnerId())
                .requestedBy(TokenContext.get()).createdAt(now).build());
        r.setMessage(trimToNull(req.message()));
        r.setStatus("pending");
        r.setRequestedAt(now);
        r.setCompletedAt(null);
        r.setUpdatedAt(now);
        requestRepository.save(r);

        boolean share = Boolean.TRUE.equals(req.shareTranscript());
        if (share && !session.isShareTranscriptWithCoach()) {
            session.setShareTranscriptWithCoach(true);
            session.setUpdatedAt(now);
            sessionRepository.save(session);
            audit.record("sales.review.share", "review_session", sessionId, Map.of("share", true));
        }
        audit.record("sales.coach.request", "review_coach_request", r.getId(),
            Map.of("sessionId", sessionId.toString(), "shareTranscript", share));

        notificationService.notifyUser(session.getCoachOwnerId(), session.getCompanyId(),
            "sales_coach_request", "有人请你补充复盘",
            (nameOf(session.getReviewerId()) + " 请你看一下「"
                + (session.getCustomerName() == null ? "本次对话" : session.getCustomerName()) + "」的复盘。"),
            "/sales/coach/requests/" + sessionId, "review_coach_request", r.getId(),
            "sales:coach-request:" + r.getId() + ":" + now.toEpochSecond(java.time.ZoneOffset.UTC));
        return supplementView(sessionId);
    }

    @Transactional
    public void cancel(UUID sessionId) {
        ReviewSession session = access.requireSession(sessionId);
        if (!session.getReviewerId().equals(TokenContext.get())) {
            throw new BusinessException(403, "只有发起人能找到这条请求");
        }
        ReviewCoachRequest r = requestRepository.findBySessionId(sessionId)
            .orElseThrow(() -> new BusinessException(404, "没有待补充请求"));
        if (!"pending".equals(r.getStatus())) return;
        r.setStatus("cancelled");
        r.setUpdatedAt(LocalDateTime.now());
        requestRepository.save(r);
        audit.record("sales.coach.cancel", "review_coach_request", r.getId(),
            Map.of("sessionId", sessionId.toString()));
    }

    /** 发起人/管理员看自己会话上的补充 */
    public CoachSupplementView supplementView(UUID sessionId) {
        access.requireSession(sessionId);
        ReviewCoachRequest r = requestRepository.findBySessionId(sessionId).orElse(null);
        List<CoachNoteView> notes = noteRepository.findBySessionIdOrderByCreatedAtAsc(sessionId).stream()
            .map(n -> new CoachNoteView(n.getId(), nameOf(n.getCoachUserId()), n.getContent(), n.getCreatedAt()))
            .toList();
        return new CoachSupplementView(r == null ? null : r.getStatus(),
            r == null ? null : r.getMessage(), notes);
    }

    /** 教练：待我补充/已补充列表 */
    public Page<CoachRequestView> listMine(int page, int size, String status) {
        UUID me = TokenContext.get();
        UUID companyId = access.requireCompanyId();
        Pageable p = PageRequest.of(Math.max(page, 1) - 1, Math.min(Math.max(size, 1), 100));
        String st = trimToNull(status);
        Page<ReviewCoachRequest> rows = st == null
            ? requestRepository.findByCompanyIdAndCoachUserIdOrderByRequestedAtDesc(companyId, me, p)
            : requestRepository.findByCompanyIdAndCoachUserIdAndStatusOrderByRequestedAtDesc(companyId, me, st, p);
        List<UUID> sessionIds = rows.getContent().stream().map(ReviewCoachRequest::getSessionId).toList();
        Map<UUID, ReviewSession> sessions = new HashMap<>();
        sessionRepository.findAllById(sessionIds).forEach(s -> sessions.put(s.getId(), s));
        Map<UUID, String> names = namesOf(rows.getContent().stream()
            .map(ReviewCoachRequest::getRequestedBy).distinct().toList());
        return rows.map(r -> {
            ReviewSession s = sessions.get(r.getSessionId());
            return new CoachRequestView(r.getSessionId(),
                s == null ? null : names.get(s.getReviewerId()),
                s == null ? null : s.getCustomerName(),
                s == null ? null : s.getTopic(),
                r.getMessage(), r.getStatus(), r.getRequestedAt(), r.getCompletedAt());
        });
    }

    /** 教练工作台：报告只读；原文只有授权才返回 */
    public CoachWorkspaceView workspace(UUID sessionId) {
        UUID me = TokenContext.get();
        ReviewSession s = sessionRepository.findById(sessionId)
            .filter(x -> x.getCompanyId().equals(access.requireCompanyId()))
            .orElseThrow(() -> new BusinessException(404, "复盘不存在或不可访问"));
        ReviewCoachRequest r = requestRepository.findBySessionId(sessionId).orElse(null);
        boolean self = s.getCoachOwnerId().equals(me) && s.getReviewerId().equals(me);
        if (!s.getCoachOwnerId().equals(me) || (r == null && !self)) {
            throw new BusinessException(404, "复盘不存在或不可访问");
        }
        ReviewReport latest = reportRepository.findFirstBySessionIdOrderByVersionDesc(sessionId).orElse(null);
        List<ReviewSegmentView> segments = s.isShareTranscriptWithCoach()
            ? segmentRepository.findBySessionIdOrderBySeqAsc(sessionId).stream()
                .map(x -> new ReviewSegmentView(x.getId(), x.getSeq(), x.getSpeaker(), x.getText()))
                .toList()
            : List.of();
        List<CoachNoteView> notes = noteRepository.findBySessionIdOrderByCreatedAtAsc(sessionId).stream()
            .map(n -> new CoachNoteView(n.getId(), nameOf(n.getCoachUserId()), n.getContent(), n.getCreatedAt()))
            .toList();
        return new CoachWorkspaceView(sessionId, nameOf(s.getReviewerId()),
            s.getCustomerName(), s.getTopic(), s.isShareTranscriptWithCoach(),
            latest == null ? null : toReportView(latest), segments, notes,
            r == null ? null : r.getStatus(), r == null ? null : r.getMessage());
    }

    @Transactional
    public CoachNoteView addNote(UUID sessionId, String content) {
        UUID me = TokenContext.get();
        ReviewSession s = sessionRepository.findById(sessionId)
            .filter(x -> x.getCompanyId().equals(access.requireCompanyId()))
            .orElseThrow(() -> new BusinessException(404, "复盘不存在或不可访问"));
        ReviewCoachRequest r = requestRepository.findBySessionId(sessionId).orElse(null);
        boolean self = s.getCoachOwnerId().equals(me) && s.getReviewerId().equals(me);
        if (!s.getCoachOwnerId().equals(me) || (r == null && !self)) {
            throw new BusinessException(404, "复盘不存在或不可访问");
        }
        if ("closed".equals(s.getStatus())) {
            throw new BusinessException(409, "复盘已关闭");
        }
        String text = content == null ? "" : content.trim();
        if (text.isEmpty()) throw new BusinessException(400, "补充内容不能为空");

        LocalDateTime now = LocalDateTime.now();
        ReviewCoachNote note = noteRepository.save(ReviewCoachNote.builder()
            .id(UUID.randomUUID()).companyId(s.getCompanyId()).sessionId(sessionId)
            .coachUserId(me).content(text).createdAt(now).build());

        if (r != null && "pending".equals(r.getStatus())) {
            r.setStatus("done");
            r.setCompletedAt(now);
            r.setUpdatedAt(now);
            requestRepository.save(r);
            if (!s.getReviewerId().equals(me)) {
                notificationService.notifyUser(s.getReviewerId(), s.getCompanyId(),
                    "sales_coach_note", "教练补充了复盘",
                    nameOf(me) + " 补充了「"
                        + (s.getCustomerName() == null ? "本次对话" : s.getCustomerName()) + "」的复盘。",
                    "/sales/reviews/" + sessionId, "review_coach_note", note.getId(),
                    "sales:coach-note:" + note.getId());
            }
        }
        audit.record("sales.coach.note", "review_coach_note", note.getId(),
            Map.of("sessionId", sessionId.toString(), "chars", text.length()));
        return new CoachNoteView(note.getId(), nameOf(me), text, now);
    }

    private ReviewReportView toReportView(ReviewReport r) {
        JsonNode payload = null;
        if (r.getPayload() != null && !r.getPayload().isBlank()) {
            try {
                payload = objectMapper.readTree(r.getPayload());
            } catch (Exception ignored) { /* payload 坏了不影响补充 */ }
        }
        return new ReviewReportView(r.getId(), r.getVersion(), r.getStatus(), r.getSummary(),
            payload, r.getErrorMessage(), r.getModelName(), r.getGeneratedAt());
    }

    private String nameOf(UUID userId) {
        if (userId == null) return "同事";
        return userRepository.findById(userId).map(User::getName).orElse("同事");
    }

    private Map<UUID, String> namesOf(List<UUID> ids) {
        if (ids.isEmpty()) return Map.of();
        Map<UUID, String> out = new HashMap<>();
        userRepository.findAllById(ids).forEach(u -> out.put(u.getId(), u.getName()));
        return out;
    }

    private String trimToNull(String s) {
        if (s == null) return null;
        String t = s.trim();
        return t.isEmpty() ? null : t;
    }
}
```

### E2.1 测试中发现的越权问题（已修）

`POST /auth/register` 在 SecurityConfig 里是 `permitAll`，但 `AuthController` 只挡了 `super_admin`，直接信任请求体 `role`。匿名请求只要知道 companyId，就能注册成 `company_admin`，实测拿到 `USER_MANAGE`、`AI_MODEL_MANAGE`，可读全公司用户与企业 Key 配置。`/auth/register/with-code` 也允许请求体覆盖注册码的 `defaultRole`，员工码可提权。

修复：

1. 新增 `service/RegistrationPolicy`（纯策略，可单测）：匿名只能创建 `employee`；创建 `company_admin` 需要调用方有 `USER_MANAGE` 且目标企业等于调用方企业；`super_admin` 仍只允许 `super_admin` 创建。
2. `/auth/register` 解析调用方 token 后调用策略，不再只判断 `super_admin`。
3. `/auth/register/with-code` 只使用注册码的 `defaultRole`，忽略请求体 `role`。

实机复验：匿名建 company_admin 403、匿名建 employee 200；super_admin 建 company_admin 200；company_admin 同企业建 company_admin 200；员工建 company_admin 403；company_admin 跨企业建员工 403；员工码 + 请求体 `role=company_admin` 最终仍是 `employee`。测试账号/注册码已清理。
